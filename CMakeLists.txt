cmake_minimum_required(VERSION 3.24)
project(strix_trellis VERSION 0.3.0 LANGUAGES C CXX)

option(STRIX_TRELLIS_ENABLE_HIP "Build available HIP programs" OFF)
option(STRIX_TRELLIS_BUILD_TESTING "Build CPU contract checks" ON)
option(STRIX_TRELLIS_BUILD_BENCHMARKS "Build optional HIP benchmark programs" OFF)
option(STRIX_TRELLIS_BUILD_EXPERIMENTAL_K2_LOCAL "Build opt-in K2 intra-block reduction qualification" OFF)
# Pretransformed expert pipeline (STRIX_TRELLIS_TP_MOE_PIPELINED_K2, OFF by default).
# The defaults below are its qualified configuration: bitwise equal to the S4
# operator, and as a whole measured faster than the pipeline with these options
# off in every measured case (docs/experimental-k2-lean.md). They have no effect
# without the pipeline; set any of them back explicitly to reproduce older builds.
set(STRIX_TRELLIS_TP_MOE_PREFETCH_ROWS "4" CACHE STRING "Multirow activation lookahead: 0, 1, 2 or 4 tiles")
if(NOT STRIX_TRELLIS_TP_MOE_PREFETCH_ROWS MATCHES "^(0|1|2|4)$")
  message(FATAL_ERROR "STRIX_TRELLIS_TP_MOE_PREFETCH_ROWS must be 0, 1, 2 or 4")
endif()
set(STRIX_TRELLIS_TP_MOE_CORE_WAVES "4" CACHE STRING "Waves per expert pipeline core CTA: 2, 4 or 8")
if(NOT STRIX_TRELLIS_TP_MOE_CORE_WAVES MATCHES "^(2|4|8)$")
  message(FATAL_ERROR "STRIX_TRELLIS_TP_MOE_CORE_WAVES must be 2, 4 or 8")
endif()
set(STRIX_TRELLIS_TP_MOE_SINGLE_DEPTH "8" CACHE STRING "Single-row weight/activation lookahead: 1, 2, 4, 8 or 16")
if(NOT STRIX_TRELLIS_TP_MOE_SINGLE_DEPTH MATCHES "^(1|2|4|8|16)$")
  message(FATAL_ERROR "STRIX_TRELLIS_TP_MOE_SINGLE_DEPTH must be 1, 2, 4, 8 or 16")
endif()
set(STRIX_TRELLIS_TP_MOE_DOWN_DEPTH "6" CACHE STRING "Down single-row lookahead: 0 inherits the shared depth; otherwise 1, 2, 4, 6, 8, 9, 12 or 16")
if(NOT STRIX_TRELLIS_TP_MOE_DOWN_DEPTH MATCHES "^(0|1|2|4|6|8|9|12|16)$")
  message(FATAL_ERROR "STRIX_TRELLIS_TP_MOE_DOWN_DEPTH must be 0, 1, 2, 4, 6, 8, 9, 12 or 16")
endif()
set(STRIX_TRELLIS_TP_MOE_ROW_CAP "2" CACHE STRING "Maximum rows decoded together: 0, 2 or 4")
if(NOT STRIX_TRELLIS_TP_MOE_ROW_CAP MATCHES "^(0|2|4)$")
  message(FATAL_ERROR "STRIX_TRELLIS_TP_MOE_ROW_CAP must be 0, 2 or 4")
endif()
option(STRIX_TRELLIS_TP_MOE_HASH32 "Use explicit modulo32 multiply for the unchanged K2 state hash" OFF)
option(STRIX_TRELLIS_TP_MOE_FOLD_R1 "Submit R1 split plans as two launches with in-core prepare and last-CTA finishes (same bits)" OFF)
set(STRIX_TRELLIS_TP_MOE_FOLD_PREPARE "0" CACHE STRING "Folded R1: prepare in each gate CTA prologue (1) or one prepare launch (0)")
set(STRIX_TRELLIS_TP_MOE_FOLD_GATE_FINISH "1" CACHE STRING "Folded R1: gate finish in the last CTA (1) or its own launch (0)")
option(STRIX_TRELLIS_TP_MOE_DECLARED_SWIGLU "Expert SwiGLU with the DeepSeek V4.1 clamp (10) and a declared exp reproducible on the host (changes H)" OFF)
option(STRIX_TRELLIS_TP_MOE_LEAN_CORE "Single-row S4 core with uniform activation loads and select-free windows (same bits)" ON)
option(STRIX_TRELLIS_TP_MOE_ADAPTIVE_ROWS "Separate one/two-row and shared-expert pipeline kernels" ON)
option(STRIX_TRELLIS_TP_MOE_PARALLEL_FINISH "Finish down routes in parallel before the ordered slot combine" OFF)
option(STRIX_TRELLIS_TP_MOE_GROUPED_FINISH "Finish all six down routes and combine them in one CTA" ON)
option(STRIX_TRELLIS_TP_MOE_SPLIT_PREPARE "Prepare gate and up projections in separate CTAs" ON)
option(STRIX_TRELLIS_TP_MOE_SPLIT_GATE_FINISH "Transform gate and up concurrently in two waves" ON)
set(STRIX_TRELLIS_TP_MOE_GATE_TILES "1" CACHE STRING "H128 tiles per split gate epilogue CTA: 1, 2 or 4")
if(NOT STRIX_TRELLIS_TP_MOE_GATE_TILES MATCHES "^(1|2|4)$")
  message(FATAL_ERROR "STRIX_TRELLIS_TP_MOE_GATE_TILES must be 1, 2 or 4")
endif()
if(NOT STRIX_TRELLIS_TP_MOE_GATE_TILES STREQUAL "1" AND NOT STRIX_TRELLIS_TP_MOE_SPLIT_GATE_FINISH)
  message(FATAL_ERROR "Gate tile grouping requires split gate finish")
endif()
option(STRIX_TRELLIS_TP_MOE_SINGLE_ROW "Schedule each selected route independently in the pipeline" OFF)
option(STRIX_TRELLIS_TP_MOE_PREPARE_ROUTES "Resolve owner maps once per pipeline stage" OFF)
option(STRIX_TRELLIS_TP_MOE_FUSE_DOWN_PREPARE "Fuse down input transform into the full pipeline gate epilogue" ON)
option(STRIX_TRELLIS_TP_MOE_PIPELINED_K2 "Use pretransformed S4 expert pipeline in explicit split plans" OFF)
option(STRIX_TRELLIS_TP_MOE_A8R "Build explicit K2 A8R plans with a distinct arithmetic identity" OFF)
option(STRIX_TRELLIS_TP_MOE_A8R_WIDE "Assign complete integer output pairs to lanes in explicit A8R plans" OFF)
set(STRIX_TRELLIS_TP_MOE_A8R_OUTPUTS_PER_LANE "2" CACHE STRING "Complete outputs per wide A8R lane: 1 or 2")
if(NOT STRIX_TRELLIS_TP_MOE_A8R_OUTPUTS_PER_LANE MATCHES "^(1|2)$")
  message(FATAL_ERROR "A8R outputs per lane must be 1 or 2")
endif()
set(STRIX_TRELLIS_TP_MOE_A8R_CLUSTER_UNROLL "1" CACHE STRING "Wide A8R hash clusters per unrolled iteration: 1, 2 or 4")
if(NOT STRIX_TRELLIS_TP_MOE_A8R_CLUSTER_UNROLL MATCHES "^(1|2|4)$")
  message(FATAL_ERROR "A8R cluster unroll must be 1, 2 or 4")
endif()
option(STRIX_TRELLIS_TP_MOE_A8R_SCALAR_INPUT "Use wave-uniform activation scalars in wide A8R kernels" OFF)
option(STRIX_TRELLIS_TP_MOE_A8R_PREPARE_AFFINE "Precompute integer affine correction and exact scale exponent in A8R prepare" OFF)
if(STRIX_TRELLIS_TP_MOE_A8R_PREPARE_AFFINE AND NOT STRIX_TRELLIS_TP_MOE_A8R)
  message(FATAL_ERROR "Prepared affine metadata requires explicit A8R")
endif()
set(STRIX_TRELLIS_TP_MOE_A8R_HASH_FENCE "0" CACHE STRING "Wide A8R hash lifetime boundary: 0, 1, 2 or 4 clusters")
if(NOT STRIX_TRELLIS_TP_MOE_A8R_HASH_FENCE MATCHES "^(0|1|2|4)$")
  message(FATAL_ERROR "A8R hash fence must be 0, 1, 2 or 4")
endif()
if(STRIX_TRELLIS_TP_MOE_A8R_SCALAR_INPUT AND NOT STRIX_TRELLIS_TP_MOE_A8R_WIDE)
  message(FATAL_ERROR "Scalar A8R input scheduling requires wide A8R")
endif()
if(STRIX_TRELLIS_TP_MOE_A8R_WIDE AND NOT STRIX_TRELLIS_TP_MOE_A8R)
  message(FATAL_ERROR "A8R wide scheduling requires the explicit A8R identity")
endif()
set(STRIX_TRELLIS_TP_MOE_PERSISTENT_GRID "0" CACHE STRING "Cooperative grid override, 0 selects one CTA per compute unit")
if(NOT STRIX_TRELLIS_TP_MOE_PERSISTENT_GRID MATCHES "^(0|[1-9][0-9]*)$" OR STRIX_TRELLIS_TP_MOE_PERSISTENT_GRID GREATER 512)
  message(FATAL_ERROR "Persistent grid must be an integer in 0..512")
endif()
option(STRIX_TRELLIS_TP_MOE_PERSISTENT "Submit the full pipeline as one occupancy-admitted cooperative kernel" OFF)
if(STRIX_TRELLIS_TP_MOE_PERSISTENT AND NOT STRIX_TRELLIS_TP_MOE_CORE_WAVES STREQUAL "4")
  message(FATAL_ERROR "Persistent MoE currently uses four-wave logical core tiles")
endif()
if(STRIX_TRELLIS_TP_MOE_PERSISTENT AND STRIX_TRELLIS_TP_MOE_SINGLE_ROW)
  message(FATAL_ERROR "Persistent MoE uses grouped owner maps; the independent route experiment is separate")
endif()
if(STRIX_TRELLIS_TP_MOE_PERSISTENT AND NOT (STRIX_TRELLIS_TP_MOE_PIPELINED_K2 AND STRIX_TRELLIS_TP_MOE_FUSE_DOWN_PREPARE AND STRIX_TRELLIS_TP_MOE_PREPARE_ROUTES))
  message(FATAL_ERROR "Persistent MoE requires pipeline, fused down prepare and prepared routes")
endif()
option(STRIX_TRELLIS_ENABLE_EXPERIMENTAL_TP "Build explicit fused dense/expert tensor-parallel APIs" OFF)
if(STRIX_TRELLIS_TP_MOE_A8R AND (NOT STRIX_TRELLIS_ENABLE_EXPERIMENTAL_TP OR NOT STRIX_TRELLIS_TP_MOE_PIPELINED_K2 OR STRIX_TRELLIS_TP_MOE_PERSISTENT))
  message(FATAL_ERROR "A8R requires experimental TP and pipeline, without cooperative scheduling")
endif()
if(STRIX_TRELLIS_TP_MOE_PARALLEL_FINISH AND (NOT STRIX_TRELLIS_TP_MOE_PIPELINED_K2 OR STRIX_TRELLIS_TP_MOE_PERSISTENT))
  message(FATAL_ERROR "Parallel finish requires pipeline without cooperative scheduling")
endif()
# Options scoped to the pipeline are ignored without it, so they are checked only with it.
if(STRIX_TRELLIS_TP_MOE_PIPELINED_K2 AND STRIX_TRELLIS_TP_MOE_GROUPED_FINISH AND (STRIX_TRELLIS_TP_MOE_PERSISTENT OR STRIX_TRELLIS_TP_MOE_PARALLEL_FINISH))
  message(FATAL_ERROR "Grouped finish cannot be combined with cooperative or separate parallel finish; set STRIX_TRELLIS_TP_MOE_GROUPED_FINISH=OFF")
endif()
if(STRIX_TRELLIS_TP_MOE_PIPELINED_K2 AND (STRIX_TRELLIS_TP_MOE_SPLIT_PREPARE OR STRIX_TRELLIS_TP_MOE_SPLIT_GATE_FINISH) AND STRIX_TRELLIS_TP_MOE_PERSISTENT)
  message(FATAL_ERROR "Split projection preparation and epilogue require the non-cooperative pipeline; set SPLIT_PREPARE and SPLIT_GATE_FINISH OFF")
endif()
if(STRIX_TRELLIS_TP_MOE_PIPELINED_K2 AND STRIX_TRELLIS_TP_MOE_ADAPTIVE_ROWS AND (NOT STRIX_TRELLIS_TP_MOE_ROW_CAP STREQUAL "2" OR STRIX_TRELLIS_TP_MOE_SINGLE_ROW OR STRIX_TRELLIS_TP_MOE_PERSISTENT))
  message(FATAL_ERROR "Adaptive rows requires row cap 2, without independent-route or cooperative scheduling; set STRIX_TRELLIS_TP_MOE_ADAPTIVE_ROWS=OFF")
endif()

add_library(strix_trellis_api INTERFACE)
add_library(strix_trellis::api ALIAS strix_trellis_api)
target_include_directories(strix_trellis_api INTERFACE
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_compile_features(strix_trellis_api INTERFACE c_std_11 cxx_std_17)

if(STRIX_TRELLIS_ENABLE_HIP)
  # Set before enabling HIP so configuration never selects the build host GPU.
  if(NOT DEFINED CMAKE_HIP_ARCHITECTURES)
    set(CMAKE_HIP_ARCHITECTURES gfx1151 CACHE STRING "HIP target architectures")
  endif()
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 17)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/backend.hip")
    find_package(hip CONFIG REQUIRED)
    add_library(strix_trellis STATIC src/api.hip src/backend.hip src/gemm.hip src/gemv.hip src/gemv_lds.hip src/gemv_k2.hip src/gemv_fused.hip src/gemv_output_direct.hip src/gemv_grouped_direct.hip src/gemv_k2_multirow.hip src/wmma_packed.hip src/gemv_fused_both.hip)
    add_library(strix_trellis::runtime ALIAS strix_trellis)
    target_include_directories(strix_trellis PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
    target_compile_options(strix_trellis PRIVATE -O3 -ffp-contract=off)
    target_link_libraries(strix_trellis PUBLIC hip::host)
    if(STRIX_TRELLIS_ENABLE_EXPERIMENTAL_TP)
      target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_ENABLE_EXPERIMENTAL_TP=1)
      target_sources(strix_trellis PRIVATE src/tp_dense.hip src/tp_moe.hip src/tp_moe_split.hip)
      if(STRIX_TRELLIS_TP_MOE_A8R)
        target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_A8R=1)
        if(STRIX_TRELLIS_TP_MOE_A8R_PREPARE_AFFINE)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_A8R_PREPARE_AFFINE=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_A8R_WIDE)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_A8R_WIDE=1
            STRIX_TRELLIS_TP_MOE_A8R_OUTPUTS_PER_LANE=${STRIX_TRELLIS_TP_MOE_A8R_OUTPUTS_PER_LANE}
            STRIX_TRELLIS_TP_MOE_A8R_CLUSTER_UNROLL=${STRIX_TRELLIS_TP_MOE_A8R_CLUSTER_UNROLL}
            STRIX_TRELLIS_TP_MOE_A8R_HASH_FENCE=${STRIX_TRELLIS_TP_MOE_A8R_HASH_FENCE})
          if(STRIX_TRELLIS_TP_MOE_A8R_SCALAR_INPUT)
            target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_A8R_SCALAR_INPUT=1)
          endif()
        endif()
      endif()
      if(STRIX_TRELLIS_TP_MOE_PIPELINED_K2)
        target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_PIPELINED_K2=1
          STRIX_TRELLIS_TP_MOE_PREFETCH_ROWS=${STRIX_TRELLIS_TP_MOE_PREFETCH_ROWS}
          STRIX_TRELLIS_TP_MOE_SINGLE_DEPTH=${STRIX_TRELLIS_TP_MOE_SINGLE_DEPTH}
          STRIX_TRELLIS_TP_MOE_DOWN_DEPTH=${STRIX_TRELLIS_TP_MOE_DOWN_DEPTH}
          STRIX_TRELLIS_TP_MOE_CORE_WAVES=${STRIX_TRELLIS_TP_MOE_CORE_WAVES}
          STRIX_TRELLIS_TP_MOE_ROW_CAP=${STRIX_TRELLIS_TP_MOE_ROW_CAP})
        if(STRIX_TRELLIS_TP_MOE_HASH32)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_HASH32=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_DECLARED_SWIGLU)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_DECLARED_SWIGLU=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_LEAN_CORE)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_LEAN_CORE=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_FOLD_R1)
          if(NOT STRIX_TRELLIS_TP_MOE_LEAN_CORE OR NOT STRIX_TRELLIS_TP_MOE_FUSE_DOWN_PREPARE OR STRIX_TRELLIS_TP_MOE_PERSISTENT OR NOT STRIX_TRELLIS_TP_MOE_CORE_WAVES STREQUAL "4")
            message(FATAL_ERROR "STRIX_TRELLIS_TP_MOE_FOLD_R1 requires LEAN_CORE, FUSE_DOWN_PREPARE, four core waves and no PERSISTENT")
          endif()
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_FOLD_R1=1
            STRIX_TRELLIS_TP_MOE_FOLD_PREPARE=${STRIX_TRELLIS_TP_MOE_FOLD_PREPARE}
            STRIX_TRELLIS_TP_MOE_FOLD_GATE_FINISH=${STRIX_TRELLIS_TP_MOE_FOLD_GATE_FINISH})
        endif()
        if(STRIX_TRELLIS_TP_MOE_ADAPTIVE_ROWS)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_ADAPTIVE_ROWS=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_PARALLEL_FINISH)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_PARALLEL_FINISH=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_GROUPED_FINISH)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_GROUPED_FINISH=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_SPLIT_PREPARE)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_SPLIT_PREPARE=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_SPLIT_GATE_FINISH)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_SPLIT_GATE_FINISH=1
            STRIX_TRELLIS_TP_MOE_GATE_TILES=${STRIX_TRELLIS_TP_MOE_GATE_TILES})
        endif()
        if(STRIX_TRELLIS_TP_MOE_SINGLE_ROW)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_SINGLE_ROW=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_PERSISTENT)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_PERSISTENT=1
            STRIX_TRELLIS_TP_MOE_PERSISTENT_GRID=${STRIX_TRELLIS_TP_MOE_PERSISTENT_GRID})
        endif()
        if(STRIX_TRELLIS_TP_MOE_PREPARE_ROUTES)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_PREPARE_ROUTES=1)
        endif()
        if(STRIX_TRELLIS_TP_MOE_FUSE_DOWN_PREPARE)
          target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_TP_MOE_FUSE_DOWN_PREPARE=1)
        endif()
      endif()
      add_executable(strix_trellis_rank_fused_probe tests/rank_fused_probe.hip)
      target_compile_options(strix_trellis_rank_fused_probe PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_rank_fused_probe PRIVATE strix_trellis::runtime)
    endif()
    add_executable(strix_trellis_prepared_api_probe tests/prepared_api_probe.hip)
    target_compile_options(strix_trellis_prepared_api_probe PRIVATE -O3 -ffp-contract=off)
    target_link_libraries(strix_trellis_prepared_api_probe PRIVATE strix_trellis::runtime)
    if(STRIX_TRELLIS_BUILD_EXPERIMENTAL_K2_LOCAL)
      add_library(strix_trellis_k2_local STATIC src/gemv_k2_local.hip)
      target_compile_options(strix_trellis_k2_local PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_k2_local PUBLIC strix_trellis::runtime)
      add_executable(strix_trellis_k2_local_probe tests/k2_local_probe.hip)
      target_compile_options(strix_trellis_k2_local_probe PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_k2_local_probe PRIVATE strix_trellis_k2_local)
      # Qualification is explicitly invoked through the site's GPU turn runner.
    endif()
    if(STRIX_TRELLIS_BUILD_BENCHMARKS)
      add_executable(strix_trellis_api_linear_bench bench/api_linear_bench.hip)
      target_compile_options(strix_trellis_api_linear_bench PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_api_linear_bench PRIVATE strix_trellis)
      add_executable(strix_trellis_graph_linear_bench bench/graph_linear_bench.hip)
      target_compile_options(strix_trellis_graph_linear_bench PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_graph_linear_bench PRIVATE strix_trellis)
      add_executable(strix_trellis_head_linear_bench bench/head_linear_bench.hip)
      target_compile_options(strix_trellis_head_linear_bench PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_head_linear_bench PRIVATE strix_trellis)
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/gpu_runtime.hip")
      add_executable(strix_trellis_gpu_runtime tests/gpu_runtime.hip)
      target_compile_options(strix_trellis_gpu_runtime PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_gpu_runtime PRIVATE strix_trellis)

  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/microbench.hip")
    add_executable(strix_trellis_microbench src/microbench.hip)
    target_link_libraries(strix_trellis_microbench PRIVATE strix_trellis::api)
    # GPU programs are deliberately not registered with CTest.
  else()
    message(STATUS "No src/microbench.hip yet; HIP configured without GPU targets")
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/design_probe.hip")
    add_executable(strix_trellis_design_probe src/design_probe.hip)
    target_link_libraries(strix_trellis_design_probe PRIVATE strix_trellis::api)
