cmake_minimum_required(VERSION 3.24)
project(strix_trellis VERSION 0.3.0 LANGUAGES C CXX)

option(STRIX_TRELLIS_ENABLE_HIP "Build available HIP programs" OFF)
option(STRIX_TRELLIS_BUILD_TESTING "Build CPU contract checks" ON)
option(STRIX_TRELLIS_BUILD_BENCHMARKS "Build optional HIP benchmark programs" OFF)
option(STRIX_TRELLIS_BUILD_EXPERIMENTAL_K2_LOCAL "Build opt-in K2 intra-block reduction qualification" OFF)
option(STRIX_TRELLIS_ENABLE_EXPERIMENTAL_TP "Build explicit fused dense/expert tensor-parallel APIs" OFF)

add_library(strix_trellis_api INTERFACE)
add_library(strix_trellis::api ALIAS strix_trellis_api)
target_include_directories(strix_trellis_api INTERFACE
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_compile_features(strix_trellis_api INTERFACE c_std_11 cxx_std_17)

if(STRIX_TRELLIS_ENABLE_HIP)
  # Set before enabling HIP so configuration never selects the build host GPU.
  if(NOT DEFINED CMAKE_HIP_ARCHITECTURES)
    set(CMAKE_HIP_ARCHITECTURES gfx1151 CACHE STRING "HIP target architectures")
  endif()
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 17)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/backend.hip")
    find_package(hip CONFIG REQUIRED)
    add_library(strix_trellis STATIC src/api.hip src/backend.hip src/gemm.hip src/gemv.hip src/gemv_lds.hip src/gemv_k2.hip src/gemv_fused.hip src/gemv_output_direct.hip src/gemv_grouped_direct.hip src/gemv_k2_multirow.hip src/wmma_packed.hip src/gemv_fused_both.hip)
    add_library(strix_trellis::runtime ALIAS strix_trellis)
    target_include_directories(strix_trellis PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
    target_compile_options(strix_trellis PRIVATE -O3 -ffp-contract=off)
    target_link_libraries(strix_trellis PUBLIC hip::host)
    if(STRIX_TRELLIS_ENABLE_EXPERIMENTAL_TP)
      target_compile_definitions(strix_trellis PRIVATE STRIX_TRELLIS_ENABLE_EXPERIMENTAL_TP=1)
      target_sources(strix_trellis PRIVATE src/tp_dense.hip src/tp_moe.hip)
      add_executable(strix_trellis_rank_fused_probe tests/rank_fused_probe.hip)
      target_compile_options(strix_trellis_rank_fused_probe PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_rank_fused_probe PRIVATE strix_trellis::runtime)
    endif()
    add_executable(strix_trellis_prepared_api_probe tests/prepared_api_probe.hip)
    target_compile_options(strix_trellis_prepared_api_probe PRIVATE -O3 -ffp-contract=off)
    target_link_libraries(strix_trellis_prepared_api_probe PRIVATE strix_trellis::runtime)
    if(STRIX_TRELLIS_BUILD_EXPERIMENTAL_K2_LOCAL)
      add_library(strix_trellis_k2_local STATIC src/gemv_k2_local.hip)
      target_compile_options(strix_trellis_k2_local PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_k2_local PUBLIC strix_trellis::runtime)
      add_executable(strix_trellis_k2_local_probe tests/k2_local_probe.hip)
      target_compile_options(strix_trellis_k2_local_probe PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_k2_local_probe PRIVATE strix_trellis_k2_local)
      # Qualification is explicitly invoked through the site's GPU turn runner.
    endif()
    if(STRIX_TRELLIS_BUILD_BENCHMARKS)
      add_executable(strix_trellis_api_linear_bench bench/api_linear_bench.hip)
      target_compile_options(strix_trellis_api_linear_bench PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_api_linear_bench PRIVATE strix_trellis)
      add_executable(strix_trellis_graph_linear_bench bench/graph_linear_bench.hip)
      target_compile_options(strix_trellis_graph_linear_bench PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_graph_linear_bench PRIVATE strix_trellis)
      add_executable(strix_trellis_head_linear_bench bench/head_linear_bench.hip)
      target_compile_options(strix_trellis_head_linear_bench PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_head_linear_bench PRIVATE strix_trellis)
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/gpu_runtime.hip")
      add_executable(strix_trellis_gpu_runtime tests/gpu_runtime.hip)
      target_compile_options(strix_trellis_gpu_runtime PRIVATE -O3 -ffp-contract=off)
      target_link_libraries(strix_trellis_gpu_runtime PRIVATE strix_trellis)
    endif()
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/microbench.hip")
    add_executable(strix_trellis_microbench src/microbench.hip)
    target_link_libraries(strix_trellis_microbench PRIVATE strix_trellis::api)
    # GPU programs are deliberately not registered with CTest.
  else()
    message(STATUS "No src/microbench.hip yet; HIP configured without GPU targets")
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/design_probe.hip")
    add_executable(strix_trellis_design_probe src/design_probe.hip)
    target_link_libraries(strix_trellis_design_probe PRIVATE strix_trellis::api)
  endif()
endif()

  target_compile_options(strix_trellis_k2_local_policy PRIVATE -UNDEBUG)

  add_test(NAME api_c COMMAND strix_trellis_api_c)
  add_test(NAME api_cxx COMMAND strix_trellis_api_cxx)
  if(UNIX)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME v14b_actual_factories COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/prove_v14b_factories_cpu.py" "${CMAKE_CURRENT_BINARY_DIR}/factory-proof")
    add_test(NAME v14b_shell_scope COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_v14b_shell_scope_cpu.py" "${CMAKE_CURRENT_BINARY_DIR}/shell-scope-proof")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fake_hip/hip")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fake_hip/hip/hip_runtime.h"
      "#define STRIX_API_FAKE_HIP\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/tests/test_api_contract.cpp\"\n#undef STRIX_API_FAKE_HIP\n")
    add_executable(strix_trellis_api_contract tests/test_api_contract.cpp)
    target_include_directories(strix_trellis_api_contract PRIVATE
      "${CMAKE_CURRENT_BINARY_DIR}/fake_hip" "${CMAKE_CURRENT_SOURCE_DIR}/include")
    target_compile_options(strix_trellis_api_contract PRIVATE -UNDEBUG -Wall -Wextra -Werror -pedantic)
    add_test(NAME api_contract COMMAND strix_trellis_api_contract)

      "${CMAKE_CURRENT_BINARY_DIR}/fake_hip" "${CMAKE_CURRENT_SOURCE_DIR}/include")

    add_test(NAME tp_api_contract COMMAND strix_trellis_tp_api_contract)
    find_program(BASH_EXECUTABLE bash REQUIRED)
    add_test(NAME tooling_safety COMMAND "${BASH_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/safety.sh")
  endif()
endif()
